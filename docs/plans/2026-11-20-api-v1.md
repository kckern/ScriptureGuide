# API v1 Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Serve the v1 API on `api.scripture.guide` (versioned, path options, passages, several translations, parallel, search, random) and the fixed simple formats on `json.`, `yml.`/`yaml.`, `md.`/`markdown.` and `txt.`/`plain.`. Retire the legacy PHP API response.

**Design:** `docs/plans/2026-11-20-api-v1-design.md`. Read it first. Where this plan differs, this plan wins, and Task 1 updates the design doc to match.

**Architecture:** A new module, `apps/web/src/api/v1/`, independent of the legacy lookup pipeline (`ScriptureLookupService`, `LegacyBlockService` and the presenters copy PHP output and quirks; v1 must not). It has a pure path parser, builders that turn a parsed request into a plain data object, renderers that turn that object into JSON, YAML, Markdown, text or an HTML fragment, and one Express router that runs before the existing routes for the API hosts. Database access for v1 goes through one new repository, `ApiRepository`, so builders are tested with a fake repository.

**Tech stack:** Node 24, Express 4, ESM, vitest + supertest, mysql2, `scripture-guide` (reference parsing), `js-yaml` (new dependency for the web app).

## Changes from the design doc

Found while reading the code; Task 1 writes them into the design doc.

1. **No `osis` field.** There is no OSIS book table. `reference` entries carry `slug` (`john.3.16-18`, the site's URL form) instead.
2. **Parallel rows** are one verse group per side, not arrays: `ParallelService.build` already joins compound verses into one cell. `match` is `quote`, `allusion` or `none` (from the row's CSS class), with a separate boolean `significant`. The `html` option does not apply to parallel.
3. **`parallels` on passages** lists the source passages that `scripture_references` links to (`[{ "reference": "JST John 3:16", "slug": "jst.john.3.16" }]`), without a translation.
4. **Simple `json.`/`yml.`** always include `chapter` on each verse, so the shape is fixed. With several translations the body is a list with one object per translation.
5. **`plain.` and `markdown.`** move to the v1 renderers along with `txt.` and `md.`. Their legacy parity cases are retired. `plainFor` in `formats/plain.js` stays: `AjaxController` uses it.
6. **Main-host shortcuts:** `scripture.guide/api/...` and `/json/...` are served by the v1 router as if they came from those subdomains.
7. **HTML fragments** don't keep paragraph breaks in v1 (verse text is flattened by `plainText`).

```js
export const STATUS = Object.freeze({
  unparseable_reference: 400,
  empty_query: 400,
  parallel_not_supported: 400,
  search_not_supported: 400,
  unknown_book: 404,
  chapter_out_of_range: 404,
  verse_out_of_range: 404,
  unknown_translation: 404,
  not_in_translation: 404,
  unsupported_api_version: 404,
});

export class ApiError extends Error {
  constructor(code, message, suggestion = null) {
    super(message);
    this.code = code;
    this.status = STATUS[code] ?? 400;
    this.suggestion = suggestion;
  }

  toJSON() {
    return { code: this.code, message: this.message, ...(this.suggestion ? { suggestion: this.suggestion } : {}) };
  }
}
```

**Step 4: Write `path.js`**

```js
// The v1 URL path: /{api version}/{options...}/{translation}/{reference}, plus the parallel
// (a~b), search (/search/{scope...}/{terms}) and random (/random, /{translation}/random,
// /random/{volume}) forms. Pure: the caller says which translation ids exist.
import { ApiError } from './errors.js';

export const CURRENT_API = 1;
export const OPTIONS = Object.freeze(['html', 'ids']);
export const VOLUMES = Object.freeze(['ot', 'nt', 'bom', 'dc', 'pgp']);
export const RESERVED = Object.freeze([...OPTIONS, ...VOLUMES, 'search', 'random']);
const API_VERSION = /^v(\d+)$/i;
const PAGE = /^page-(\d+)$/i;

function decode(segment) {
  try { return decodeURIComponent(segment); } catch { return segment; }
}

// "nrsv.kjv*lds" → ['NRSV', 'KJV'] in stored case; the "*format" part is layout only
function translationList(spec, isTranslation, canonical) {
  const ids = spec.split('.').filter(Boolean).map(piece => piece.split('*')[0]);
  for (const id of ids) {
    if (!isTranslation(id)) throw new ApiError('unknown_translation', `There is no translation "${id}".`);
  }
  return ids.map(canonical);
}

export function parseApiPath(path, isTranslation, canonical = id => id.toUpperCase()) {
  const segs = String(path).split('/').filter(Boolean).map(decode);

  let api = CURRENT_API;
  const version = segs[0]?.match(API_VERSION);
  if (version) {
    api = Number(version[1]);
    segs.shift();
    if (api !== CURRENT_API) throw new ApiError('unsupported_api_version', `API version v${api} does not exist. The current version is v${CURRENT_API}.`);
  }

  const options = [];
  while (segs.length && OPTIONS.includes(segs[0].toLowerCase())) options.push(segs.shift().toLowerCase());

  const base = { api, options };
  const lower = segs.map(s => s.toLowerCase());

  if (lower[0] === 'search') return { ...base, kind: 'search', ...searchScope(segs.slice(1), isTranslation, canonical) };

  if (lower[0] === 'random') return { ...base, kind: 'random', translations: [], volume: volumeOf(lower[1]) };
  if (lower[1] === 'random') {
    return { ...base, kind: 'random', translations: translationList(segs[0], isTranslation, canonical), volume: volumeOf(lower[2]) };
  }

  const rest = segs.join('/');
  if (rest.slice(1).includes('~')) return { ...base, kind: 'parallel', query: rest };

  if (!segs.length) throw new ApiError('unparseable_reference', 'No reference was given.');
  if (segs.length === 1) return { ...base, kind: 'passage', translations: [], reference: segs[0] };
  return {
    ...base,
    kind: 'passage',
    translations: translationList(segs[0], isTranslation, canonical),
    reference: segs.slice(1).join('; '),
  };
}

function volumeOf(word) {
  return word && VOLUMES.includes(word) ? word : null;
}

// /search/{translation?}/{volume?}/{page-n?}/{terms}: terms are the last segment
function searchScope(segs, isTranslation, canonical) {
  if (!segs.length) throw new ApiError('empty_query', 'Search terms are empty.');
  const terms = segs[segs.length - 1].replace(/[+.]/g, ' ').replace(/\s+/g, ' ').trim();
  if (!terms) throw new ApiError('empty_query', 'Search terms are empty.');
  const scope = { terms, translation: null, volume: null, page: 1 };
  for (const seg of segs.slice(0, -1)) {
    const word = seg.toLowerCase();
    const page = word.match(PAGE);
    if (page) scope.page = Math.max(1, Number(page[1]));
    else if (VOLUMES.includes(word)) scope.volume = word;
    else if (isTranslation(seg)) scope.translation = canonical(seg);
    else throw new ApiError('unknown_translation', `There is no translation "${seg}".`);
  }
  return scope;
}
```

**Step 3: Write `ApiRepository.js`**

```js
import db from '../db/connection.js';

// Database reads for the v1 API (src/api/v1). Kept apart from the legacy repositories, which
// reproduce PHP queries.
export class ApiRepository {
  constructor(database = db) {
    this.db = database;
    this.translationCache = null;
  }

  // Map of lower-case id → { id, name }. Translations don't change while the process runs.
  async translations() {
    this.translationCache ??= this.db
      .query('SELECT id, title, title_long FROM scripture_versions ORDER BY priority, title')
      .then(rows => new Map(rows.map(r => [String(r.id).toLowerCase(), { id: String(r.id), name: r.title_long || r.title }])))
      .catch(err => { this.translationCache = null; throw err; });
    return this.translationCache;
  }

  // Index rows (book, chapter, verse) for verse ids, in order
  async indexRows(ids) {
    if (!ids.length) return [];
    return this.db.query(
      `SELECT v.verse_id, v.book_id, v.chapter, v.verse, b.book_title
       FROM lds_scriptures_verses v JOIN lds_scriptures_books b ON b.book_id = v.book_id
       WHERE v.verse_id IN (?) ORDER BY v.verse_id`,
      [ids]
    );
  }

  // One translation's text for verse ids
  async verses(ids, version) {
    if (!ids.length) return [];
    return this.db.query(
      'SELECT verse_id, text FROM scripture_verses WHERE version = ? AND verse_id IN (?) ORDER BY verse_id',
      [version, ids]
    );
  }

  // Section headings that start at one of the verse ids
  async headingStarts(ids, version) {
    if (!ids.length) return [];
    return this.db.query(
      `SELECT verse_id, text FROM scripture_headings
       WHERE version = ? AND type = 'heading' AND verse_id IN (?) ORDER BY verse_id, seq`,
      [version, ids]
    );
  }

  // The section heading in force at a verse: the last one at or before it in the same book
  async headingAt(verseId, bookId, version) {
    const [row] = await this.db.query(
      `SELECT h.text FROM scripture_headings h JOIN lds_scriptures_verses v ON v.verse_id = h.verse_id
       WHERE h.version = ? AND h.type = 'heading' AND v.book_id = ? AND h.verse_id <= ?
       ORDER BY h.verse_id DESC, h.seq LIMIT 1`,
      [version, bookId, verseId]
    );
    return row?.text ?? null;
  }

  // Verse ids that scripture_references names as stand-ins for these verses
  async parallelSources(ids) {
    if (!ids.length) return [];
    const rows = await this.db.query(
      'SELECT DISTINCT src_verse_id FROM scripture_references WHERE standin > 0 AND dst_verse_id IN (?) ORDER BY src_verse_id',
      [ids]
    );
    return rows.map(r => Number(r.src_verse_id));
  }

  async chapterCount(bookId) {
    const [row] = await this.db.query('SELECT MAX(chapter) AS n FROM lds_scriptures_verses WHERE book_id = ?', [bookId]);
    return Number(row?.n ?? 0);
  }

  async verseCount(bookId, chapter) {
    const [row] = await this.db.query('SELECT MAX(verse) AS n FROM lds_scriptures_verses WHERE book_id = ? AND chapter = ?', [bookId, chapter]);
    return Number(row?.n ?? 0);
  }

  // A uniformly random verse id in [min, max] that the translation has text for, or null
  async randomVerseId({ range: [min, max], version }, random = Math.random) {
    const [count] = await this.db.query(
      'SELECT COUNT(*) AS n FROM scripture_verses WHERE version = ? AND verse_id BETWEEN ? AND ?',
      [version, min, max]
    );
    const n = Number(count?.n ?? 0);
    if (!n) return null;
    const [row] = await this.db.query(
      'SELECT verse_id FROM scripture_verses WHERE version = ? AND verse_id BETWEEN ? AND ? ORDER BY verse_id LIMIT 1 OFFSET ?',
      [version, min, max, Math.floor(random() * n)]
    );
    return row ? Number(row.verse_id) : null;
  }
}

export default new ApiRepository();
```

**Step 4: Write `text.js`**

```js
// Character offsets of the <span class='searchterm'> marks that SearchService's highlighter puts
// in plain verse text, as [start, end] pairs into the text without the marks.
const OPEN = "<span class='searchterm'>";
const CLOSE = '</span>';

export function highlightOffsets(marked) {
  const out = [];
  let plain = 0;
  let i = 0;
  while (i < marked.length) {
    if (marked.startsWith(OPEN, i)) {
      const end = marked.indexOf(CLOSE, i + OPEN.length);
      const length = end - (i + OPEN.length);
      if (length > 0) out.push([plain, plain + length]);
      plain += length;
      i = end + CLOSE.length;
    } else {
      plain++;
      i++;
    }
  }
  return out;
}
```

```js
// Why a reference found no verses: an unknown book, a chapter or verse past the end, or text
// that can't be read as a reference. Returns an ApiError (the caller throws it).
import { lookupReference } from 'scripture-guide';
import { ApiError } from './errors.js';

// "jon3.16" → book "jon", chapter 3, verse 16; "2ne12" → "2ne", 12; "1 john 9:2" → "1 john", 9, 2
const PARTS = /^(\d?\s*[^\d]+?)[\s.]*(\d+)(?:[\s.:]+(\d+))?/;

export async function diagnoseMiss(query, repo) {
  const text = String(query).trim();
  const unreadable = () => new ApiError('unparseable_reference', `"${text}" can't be read as a reference.`);
  if (!/[a-z]/i.test(text) || !/\d/.test(text)) return unreadable();

  const parts = text.match(PARTS);
  if (!parts) return unreadable();
  const [, bookText, chapterText, verseText] = parts;

  const first = lookupReference(`${bookText.trim()} 1`);
  if (!first.verse_ids?.length) return new ApiError('unknown_book', `No book matches "${bookText.trim()}".`);
  const [row] = await repo.indexRows([first.verse_ids[0]]);
  if (!row) return unreadable();
  const book = row.book_title;

  const chapters = await repo.chapterCount(row.book_id);
  const chapter = Number(chapterText);
  if (chapter > chapters) {
    return new ApiError('chapter_out_of_range', `${book} has ${chapters} chapter${chapters === 1 ? '' : 's'}.`, `${book} ${chapters}`);
  }
  if (verseText) {
    const verses = await repo.verseCount(row.book_id, chapter);
    if (Number(verseText) > verses) {
      return new ApiError('verse_out_of_range', `${book} ${chapter} has ${verses} verses.`, `${book} ${chapter}:${verses}`);
    }
  }
  return unreadable();
}
```

```js
// The v1 passage object for a reference in one or more translations (design doc, "Response"
// and "Several translations").
import referenceService from '../../services/ReferenceService.js';
import apiRepository from '../../data/repositories/ApiRepository.js';
import { Reference } from '../../domain/entities/Reference.js';
import { defaultVersions } from '../../domain/rules/defaultVersions.js';
import { plainText } from '../../domain/rules/legacyText.js';
import { slugifyRef } from '../../domain/rules/legacyReference.js';
import { ApiError } from './errors.js';
import { diagnoseMiss } from './diagnose.js';
import { copyrightFor } from './copyright.js';
import { passageHtml } from './html.js';

const SITE = 'https://scripture.guide';

export function translationInfo(id, names) {
  const known = names.get(String(id).toLowerCase());
  return { id: known?.id ?? id, name: known?.name ?? id, copyright: copyrightFor(id) };
}

// The pieces of a reference ("John 3:16; 4:1; Romans 8:28"), each with its own verse ids.
// A piece without a book name takes the book before it.
async function pieces(query, { refs, repo }) {
  const whole = await refs.resolve(query);
  if (whole.isEmpty) throw await diagnoseMiss(query, repo);
  const labels = whole.normalized.split(';').map(s => s.trim()).filter(Boolean);
  if (labels.length < 2) return [{ label: whole.normalized, ids: whole.verseIds }];
  const out = [];
  let book = '';
  for (const piece of labels) {
    const label = /[a-z]/i.test(piece) ? piece : `${book} ${piece}`;
    book = new Reference({ normalized: label }).book;
    const ref = await refs.resolve(label);
    if (ref.isEmpty) throw await diagnoseMiss(label, repo);
    out.push({ label, ids: ref.verseIds });
  }
  return out;
}

function referenceEntry(label, rows) {
  const first = rows[0];
  const last = rows[rows.length - 1];
  return {
    label,
    book: first.book_title,
    slug: slugifyRef(label),
    start: { chapter: first.chapter, verse: first.verse },
    end: { chapter: last.chapter, verse: last.verse },
  };
}

// Runs of verse ids: a new run where the ids jump or where a section heading starts
function runs(ids, headingStarts) {
  const out = [];
  for (const [i, id] of ids.entries()) {
    if (i === 0 || id !== ids[i - 1] + 1 || headingStarts.has(id)) out.push([id]);
    else out[out.length - 1].push(id);
  }
  return out;
}

// The translations to use: the requested ones, or the first default (by verse range) with text
async function chooseTranslations(requested, ids, repo) {
  if (requested.length) return requested;
  for (const { version } of defaultVersions([], ids)) {
    if ((await repo.verses(ids.slice(0, 1), version)).length) return [version];
  }
  return ['LDS'];
}

export async function buildPassage({ reference, translations: requested, options = [] }, { repo = apiRepository, refs = referenceService } = {}) {
  const names = await repo.translations();
  const parts = await pieces(reference, { refs, repo });
  const allIds = parts.flatMap(p => p.ids);
  const translations = await chooseTranslations(requested, allIds, repo);
  const idsOnly = options.includes('ids');

  // Text per translation, keyed by verse id
  const texts = {};
  for (const t of translations) {
    texts[t] = new Map((await repo.verses(allIds, t)).map(r => [Number(r.verse_id), plainText(r.text)]));
  }
  if (!idsOnly && translations.every(t => texts[t].size === 0)) {
    throw new ApiError('not_in_translation', `${translations.join(', ')} has no text for ${parts.map(p => p.label).join('; ')}.`);
  }

  const rows = new Map((await repo.indexRows(allIds)).map(r => [Number(r.verse_id), r]));
  const starts = new Set();
  if (!idsOnly) for (const t of translations) for (const h of await repo.headingStarts(allIds, t)) starts.add(Number(h.verse_id));

  const passages = [];
  for (const part of parts) {
    for (const run of runs(part.ids, starts)) {
      const verses = run.map(id => {
        const r = rows.get(id);
        const verse = { id, chapter: r?.chapter ?? null, verse: r?.verse ?? null };
        if (!idsOnly) verse.texts = Object.fromEntries(translations.map(t => [t, texts[t].get(id) ?? null]));
        return verse;
      });
      const passage = {};
      if (!idsOnly) {
        passage.headings = {};
        for (const t of translations) passage.headings[t] = await repo.headingAt(run[0], rows.get(run[0])?.book_id, t);
      }
      passage.verses = verses;
      if (options.includes('html') && !idsOnly) {
        passage.html = Object.fromEntries(translations.map(t => [t, passageHtml(passage, t)]));
      }
      passages.push(passage);
    }
  }

  const parallels = [];
  for (const id of await repo.parallelSources(allIds)) {
    const ref = await refs.resolve(String(id));
    if (!ref.isEmpty) parallels.push({ reference: ref.normalized, slug: slugifyRef(ref.normalized) });
  }

  const slug = parts.map(p => slugifyRef(p.label)).join(';');
  const prefix = requested.length ? `${requested.map(t => t.toLowerCase()).join('.')}/` : '';
  return {
    api: 'v1',
    query: reference,
    translations: translations.map(t => translationInfo(t, names)),
    reference: parts.map(p => referenceEntry(p.label, p.ids.map(id => rows.get(id)).filter(Boolean))),
    passages,
    parallels,
    links: { web: `${SITE}/${prefix}${slug}` },
  };
}
```

```js
// Bodies for each host: api (full v1 JSON), and the simple formats json, yaml, markdown and
// text built from the same passage object. Content types live in CONTENT_TYPES.
import yaml from 'js-yaml';

export const CONTENT_TYPES = Object.freeze({
  api: 'application/json; charset=utf-8',
  json: 'application/json; charset=utf-8',
  yaml: 'application/yaml; charset=utf-8',
  markdown: 'text/markdown; charset=utf-8',
  text: 'text/plain; charset=utf-8',
});

function forTranslation(data, t) {
  const verses = data.passages.flatMap(p => p.verses)
    .filter(v => v.texts?.[t.id] != null)
    .map(v => ({ chapter: v.chapter, verse: v.verse, text: v.texts[t.id] }));
  return {
    reference: data.reference.map(r => r.label).join('; '),
    translation: t.id,
    text: verses.map(v => v.text).join(' '),
    verses,
    copyright: t.copyright,
  };
}

// One object for one translation, a list for several
export function simpleData(data) {
  const all = data.translations.map(t => forTranslation(data, t));
  return all.length === 1 ? all[0] : all;
}

export function renderSimple(kind, data) {
  const blocks = data.translations.map(t => forTranslation(data, t));
  switch (kind) {
    case 'json': return JSON.stringify(simpleData(data));
    case 'yaml': return yaml.dump(simpleData(data));
    case 'markdown':
      return blocks.map(b => `> ${b.text}\n>\n> — ${b.reference} (${b.translation})\n`).join('\n');
    case 'text':
      return blocks.map(b => `${b.text}\n`).join('');
    default: throw new Error(`unknown simple format ${kind}`);
  }
}

export function renderError(kind, err) {
  const error = err.toJSON();
  switch (kind) {
    case 'api': return JSON.stringify({ api: 'v1', error });
    case 'json': return JSON.stringify({ error });
    case 'yaml': return yaml.dump({ error });
    default: return `${err.message}\n`;
  }
}
```

```js
import parallelService from '../../services/ParallelService.js';
import apiRepository from '../../data/repositories/ApiRepository.js';
import { ApiError } from './errors.js';
import { translationInfo } from './passage.js';

const clean = text => String(text ?? '').replace(/\s+/g, ' ').trim();

function side(cell) {
  if (cell.verseId === '' || cell.verseId == null) return null;
  return { verse_id: Number(cell.verseId), reference: cell.ref, text: clean(cell.text) };
}

function row(r) {
  const classes = r.className.split(/\s+/);
  const match = classes.includes('quote') ? 'quote' : classes.includes('allusion') ? 'allusion' : 'none';

export async function buildParallel(query, { parallel = parallelService, repo = apiRepository } = {}) {

  const names = await repo.translations();
  return {
    api: 'v1',
    query,
    left: { translation: translationInfo(page.lver, names), reference: page.leftheader },
    right: { translation: translationInfo(page.rver, names), reference: page.rightheader },
    rows: page.rows.map(row),
```

```js
// The v1 search object: canon order, PER_PAGE results a page, match offsets instead of markup.
import searchService, { volumeRange } from '../../services/SearchService.js';
import { slugifyRef } from '../../domain/rules/legacyReference.js';
import { highlightOffsets } from './text.js';

export const PER_PAGE = 50;
const API = 'https://api.scripture.guide/v1/search';

export async function buildSearch({ terms, translation, volume, page }, { search = searchService } = {}) {
  const result = await search.search(terms, { forcedVersion: translation ?? volume ?? null });
  let rows = [...result.rows];
  if (translation && volume) {
    const [min, max] = volumeRange(volume);
    rows = rows.filter(r => r.verse_id >= min && r.verse_id <= max);
  }
  rows.sort((a, b) => a.verse_id - b.verse_id);

  const total = rows.length;
  const start = (page - 1) * PER_PAGE;
  const results = rows.slice(start, start + PER_PAGE).map(r => ({
    id: r.verse_id,
    reference: r.ref,
    slug: slugifyRef(r.refShort),
    translation: r.version,
    text: r.text,
    highlights: highlightOffsets(r.highlighted),
  }));

  const scopePath = [translation?.toLowerCase(), volume].filter(Boolean).join('/');
  const termsPath = encodeURIComponent(terms).replace(/%20/g, '+');
  const next = start + PER_PAGE < total ? `${API}/${scopePath ? `${scopePath}/` : ''}page-${page + 1}/${termsPath}` : null;
  const webVersion = translation ? `${translation.toLowerCase()}/` : '';

  return {
    api: 'v1',
    query: terms,
    scope: { translation, volume },
    total,
    truncated: !!result.truncated,
    page,
    per_page: PER_PAGE,
    results,
    links: { next, web: `https://scripture.guide/${webVersion}search/${encodeURIComponent(terms)}` },
  };
}
```

```js
// The v1 API and the simple format hosts. Runs before the site's routes; anything that isn't
// one of these hosts (or the main host's /api/ and /json/ shortcuts) falls through.
import { Router } from 'express';
import { createHash } from 'crypto';
import apiRepository from '../../data/repositories/ApiRepository.js';
import { parseApiPath } from './path.js';
import { ApiError } from './errors.js';
import { buildPassage } from './passage.js';
import { buildParallel } from './parallel.js';
import { buildSearch } from './search.js';
import { pickRandomVerse } from './random.js';
import { renderSimple, renderError, CONTENT_TYPES } from './render.js';

const HOSTS = Object.freeze({
  api: 'api', json: 'json', yml: 'yaml', yaml: 'yaml', md: 'markdown', markdown: 'markdown', txt: 'text', plain: 'text',
});
const SHORTCUTS = Object.freeze({ api: 'api', json: 'json' });
// Service paths every host answers through the site's routes
const PASS = new Set(['/health', '/healthz', '/build-info', '/favicon.ico', '/robots.txt']);

const CACHE_OK = 'public, max-age=86400, s-maxage=604800, stale-while-revalidate=86400';
const CACHE_ERROR = 'public, max-age=300';

// { kind, path } for a request this router serves, else null
export function hostKind(subdomain, path) {
  const kind = HOSTS[String(subdomain || '').toLowerCase()];
  if (kind) return { kind, path };
  const m = String(path).match(/^\/(api|json)(\/.*)$/i);
  return m ? { kind: SHORTCUTS[m[1].toLowerCase()], path: m[2] } : null;
}

function send(req, res, kind, body, cacheControl) {
  res.set('X-API-Version', '1');
  res.set('Cache-Control', cacheControl);
  res.set('Content-Type', CONTENT_TYPES[kind]);
  // A strong ETag; res.send answers 304 itself when If-None-Match matches it
  res.set('ETag', `"${createHash('sha1').update(body).digest('base64url')}"`);
  res.send(body);
}

const DEFAULTS = { repo: apiRepository, buildPassage, buildParallel, buildSearch, pickRandomVerse };

export function apiRouter(deps = DEFAULTS) {
  const router = Router();
  router.get('*', async (req, res, next) => {
    const target = hostKind(req.subdomain, req.path);
    if (!target || PASS.has(req.path)) return next();
    const { kind } = target;
    try {
      const names = await deps.repo.translations();
      const isTranslation = id => names.has(String(id).toLowerCase());
      const canonical = id => names.get(String(id).toLowerCase())?.id ?? id;
      const parsed = parseApiPath(target.path, isTranslation, canonical);

      if (kind !== 'api' && parsed.kind === 'search') throw new ApiError('search_not_supported', `Search is only on api.scripture.guide.`);
      if (kind !== 'api' && parsed.kind === 'parallel') throw new ApiError('parallel_not_supported', `Parallel is only on api.scripture.guide.`);

      let data;
      let cache = CACHE_OK;
      if (parsed.kind === 'search') data = await deps.buildSearch(parsed);
      else if (parsed.kind === 'parallel') data = await deps.buildParallel(parsed.query);
      else {
        let { reference } = parsed;
        if (parsed.kind === 'random') {
          reference = String(await deps.pickRandomVerse(parsed));
          cache = 'no-store';
        }
        data = await deps.buildPassage({ reference, translations: parsed.translations, options: kind === 'api' ? parsed.options : [] });
      }
      const body = kind === 'api' ? JSON.stringify(data) : renderSimple(kind, data);
      send(req, res, kind, body, cache);
    } catch (err) {
      if (!(err instanceof ApiError)) return next(err);
      res.status(err.status);
      send(req, res, kind, renderError(kind, err), CACHE_ERROR);
    }
  });
  return router;
}

export default apiRouter;
```
